    <script>
        // DBD Sports Products List
        const products = [
            { id: 1, name: "DBD Performance Graphic Tee", price: 3800, img: "https://images.unsplash.com/photo-1521572267360-ee0c2909d518?auto=format&fit=crop&w=600&q=80" },
            { id: 2, name: "DBD Heavyweight Street Hoodie", price: 7800, img: "https://images.unsplash.com/photo-1556905055-8f358a7a47b2?auto=format&fit=crop&w=600&q=80" },
            { id: 3, name: "DBD Sports Pro Racing Cap", price: 2900, img: "https://images.unsplash.com/photo-1588850561407-ed78c282e89b?auto=format&fit=crop&w=600&q=80" },
            { id: 4, name: "DBD Oversized Athletic Tee", price: 4500, img: "https://images.unsplash.com/photo-1503342217505-b0a15ec3261c?auto=format&fit=crop&w=600&q=80" }
        ];

        let cart = [];

        function renderProducts() {
            const grid = document.getElementById('product-grid');
            grid.innerHTML = products.map(p => `
                <div class="product-card">
                    <img src="${p.img}" class="product-img" alt="${p.name}">
                    <div class="product-info">
                        <div class="product-title">${p.name}</div>
                        <div class="product-price">LKR ${p.price.toLocaleString()}</div>
                        <select class="size-select" id="size-${p.id}">
                            <option value="Small (S)">Size: Small (S)</option>
                            <option value="Medium (M)" selected>Size: Medium (M)</option>
                            <option value="Large (L)">Size: Large (L)</option>
                            <option value="XL">Size: XL</option>
                        </select>
                        <button class="add-btn" onclick="addToCart(${p.id})">Add To Cart</button>
                    </div>
                </div>
            `).join('');
        }

        function addToCart(id) {
            const product = products.find(p => p.id === id);
            const selectedSize = document.getElementById(`size-${id}`).value;
            
            cart.push({ ...product, size: selectedSize });
            updateCart();
            toggleCart();
        }

        function updateCart() {
            document.getElementById('cart-count').innerText = cart.length;
            const itemsContainer = document.getElementById('cart-items');
            
            if(cart.length === 0) {
                itemsContainer.innerHTML = "<p style='color: var(--text-gray); margin-top:20px;'>Your cart is empty.</p>";
            } else {
                itemsContainer.innerHTML = cart.map((item, index) => `
                    <div class="cart-item">
                        <div>
                            <strong>${item.name}</strong><br>
                            <small style="color: var(--text-gray);">${item.size}</small><br>
                            <small style="color: var(--accent-red);">LKR ${item.price.toLocaleString()}</small>
                        </div>
                        <span style="color:#ef4444; font-weight:bold; cursor:pointer; font-size:20px;" onclick="removeFromCart(${index})">&times;</span>
                    </div>
                `).join('');
            }
            
            const total = cart.reduce((sum, item) => sum + item.price, 0);
            document.getElementById('cart-total').innerText = total.toLocaleString();
        }

        function removeFromCart(index) {
            cart.splice(index, 1);
            updateCart();
        }

        function toggleCart() {
            document.getElementById('cart-modal').classList.toggle('active');
        }

        function sendWhatsAppOrder() {
            if (cart.length === 0) { alert('Your cart is empty!'); return; }
            
            // ඔබේ WhatsApp දුරකථන අංකය මෙතැනට දාන්න (උදා: 9477XXXXXXX)
            let phone = "94770000000"; 
            
            let message = "Hi DBD SPORTS! I would like to place an order:%0A%0A";
            cart.forEach((item, i) => {
                message += `${i + 1}. ${item.name} - ${item.size} (LKR ${item.price.toLocaleString()})%0A`;
            });
            const total = cart.reduce((sum, item) => sum + item.price, 0);
            message += `%0A*Total Amount: LKR ${total.toLocaleString()}*`;
            
            window.open(`https://wa.me/${phone}?text=${message}`, '_blank');
        }

        renderProducts();
    </script>
